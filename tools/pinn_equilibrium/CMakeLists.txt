cmake_minimum_required(VERSION 3.16)
project(iter_pinn_equilibrium LANGUAGES CXX)
add_library(iter_pinn_equilibrium STATIC equilibrium.cpp)
target_compile_features(iter_pinn_equilibrium PUBLIC cxx_std_17)
target_include_directories(iter_pinn_equilibrium PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(iter_pinn_evaluate main.cpp)
target_link_libraries(iter_pinn_evaluate PRIVATE iter_pinn_equilibrium)
include(CTest)
if(BUILD_TESTING)
  add_executable(iter_pinn_test test_evaluator.cpp)
  target_link_libraries(iter_pinn_test PRIVATE iter_pinn_equilibrium)
  add_test(NAME iter_pinn_evaluator COMMAND iter_pinn_test)
endif()
